```662:675:scripts/video_tasks.py
def chat(backend, ollama_url, model, content, images=None, *,
         stream=False, on_delta=None, max_tokens=None) -> str:
    if (backend or 'ollama').lower() == 'mlx':
        if images:
            return mlx_vlm_chat(model, content, images, ...)   # 视觉→mlx-vlm
        return mlx_lm_chat(model, content, ...)                # 文本→mlx-lm
    return ollama_chat(ollama_url, model, content, images=images, ...)
```
